 14class QuickStockClient:
 15    """
 16    QuickStock SDK客户端类

 63    def stock_daily(self, codes: List[str], start_date: Optional[str] = None, 
 64                   end_date: Optional[str] = None, **kwargs) -> pd.DataFrame:
 65        """
 66        获取股票日线数据（同步方法）
 67        
 68        Args:
 69            codes: 股票代码列表
 70            start_date: 开始日期，格式如'2024-01-01'
 71            end_date: 结束日期，格式如'2024-01-31'
 72            
 73        Returns:
 74            包含股票日线数据的DataFrame，字段包括：
 75            - trade_date: 交易所行情日期
 76            - code: 证券代码
 77            - open: 开盘价
 78            - high: 最高价
 79            - low: 最低价
 80            - close: 收盘价
 81            - preclose: 前收盘价
 82            - volume: 成交量（股）
 83            - amount: 成交额（元）
 84            - adjustflag: 复权状态(1:后复权, 2:前复权, 3:不复权)
 85            - turn: 换手率(%)
 86            - tradestatus: 交易状态(1:正常, 0:停牌)
 87            - pctChg: 涨跌幅(%)
 88            - isST: 是否ST股(1:是, 0:否)
 89            - peTTM: 滚动市盈率
 90            - pbMRQ: 市净率
 91            - psTTM: 滚动市销率
 92            - pcfNcfTTM: 滚动市现率
 93        """
 94        return self._run_async(self.baostock.get_stock_daily(
 95            codes, start_date, end_date, **kwargs
 96        ))
