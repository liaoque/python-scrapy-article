 46        r = self._hash(n)
 47        n = self._encode(n, [3, r])

 50    def _encode(self, n, o):
 51        a = 0
 52        i = 2
 53        u = o[1]
 54        while a < len(n):
 55            o.append(n[a] ^ (u & 255))
 56            u = ~(u * 131)
 57            a += 1
 58            i += 1
 59        return o

 61    def strHash(self, userAgent):
 62        c = 0
 63        for v in userAgent:
 64            c = (c << 5) - c + ord(v)
 65            c &= 0xFFFFFFFF
 66        return c

 68    def _hash(self, n):
 69        e = 0
 70        for i in n:
 71            e = (e << 5) - e + i
 72        return e & 255

 74    def _base64(self, n):
 75        m = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_"
 76        f = []
 77        for i in range(0, len(n), 3):
 78            l = (n[i] << 16) | (n[i + 1] << 8) | n[i + 2]
 79            f.append(m[(l >> 18) & 63])
 80            f.append(m[(l >> 12) & 63])
 81            f.append(m[(l >> 6) & 63])
 82            f.append(m[l & 63])
 83        return ''.join(f)

 85    def to_buff(self, n):
 86        u = [4, 4, 4, 4, 1, 1, 1, 3, 2, 2, 2, 2, 2, 2, 2, 4, 2, 1]
 87        c = [0] * 43
 88        s = -1
 89        for v in range(len(u)):
 90            l = n[v]
 91            p = u[v]
 92            s += p
 93            d = s
 94            while p != 0:
 95                c[d] = (l & 255)
 96                l >>= 8
 97                p -= 1
 98                d -= 1
 99        return c

213    def _parse_kline_data(self, data_str, frequency):
214        """
215        解析K线数据
216        
217        Args:
218            data_str: 数据字符串
219            frequency: 频率类型
220            
221        Returns:
222            解析后的数据列表
223        """
224        result = []
225        for v in data_str.split(';'):
226            if not v:
227                continue
228            x = v.split(',')
229            if len(x) < 10:
230                continue
231            
232            def safe_float(val, default=0.0):
233                """安全转换为float"""
234                try:
235                    return float(val) if val and val.strip() else default
236                except (ValueError, TypeError):
237                    return default
238            
239            def safe_int(val, default=0):
240                """安全转换为int"""
241                try:
242                    return int(val) if val and val.strip() else default
243                except (ValueError, TypeError):
244                    return default
245            
246            try:
247                result.append({
248                    "date_at": x[0] if len(x) > 0 and x[0] else "",
249                    "start": safe_float(x[1] if len(x) > 1 else ""),
250                    "end": safe_float(x[4] if len(x) > 4 else ""),
251                    "max": safe_float(x[2] if len(x) > 2 else ""),
252                    "min": safe_float(x[3] if len(x) > 3 else ""),
253                    "count": safe_int(x[5] if len(x) > 5 else ""),
254                    "amount": safe_float(x[6] if len(x) > 6 else ""),
255                    "amplitude": safe_float(x[2] if len(x) > 2 else "") - safe_float(x[3] if len(x) > 3 else ""),
256                    "range": 0.0,
257                    "range_amount": safe_float(x[9] if len(x) > 9 else ""),
258                    "turnover_rate": 0.0
259                })
260            except Exception:
261                continue
262        
263        return self._compute_range(result)

265    def _compute_range(self, data):
266        """
267        计算涨跌幅、振幅等衍生指标
268        
269        Args:
270            data: 原始数据列表
271            
272        Returns:
273            计算后的DataFrame
274        """
275        if not data:
276            return pd.DataFrame()
277        
278        df = pd.DataFrame(data)
279        
280        # 确保数值列是float类型
281        numeric_cols = ['start', 'end', 'max', 'min', 'count', 'amount', 'amplitude', 'range', 'range_amount', 'turnover_rate']
282        for col in numeric_cols:
283            if col in df.columns:
284                df[col] = pd.to_numeric(df[col], errors='coerce').fillna(0.0)
285        
286        df2 = df.shift()
287        
288        # 计算涨跌额
289        df['range_amount'] = round(df['end'] - df2['end'], 3)
290        
291        # 计算涨跌幅
292        df['range'] = round(df['range_amount'] / df2['end'], 3)
293        
294        # 计算振幅
295        df['amplitude'] -= round((df['max'] - df2['min']) / df2['end'], 3)
296        
297        return df
